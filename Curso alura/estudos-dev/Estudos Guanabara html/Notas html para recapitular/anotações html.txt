		links

target='_blank' = ao passar esse parâmetro, quando abrir clicar no link ele ira abrir outra pagina
target='_self' = Ao clicar no link a pagina vai ser atualizada para a próxima, sem abrir outra.
rel='external' = Você esta passando que o link é externo.
rel='next' = indica que ao clicar você vai pra pagina seguinte, exemplo: você esta na pagina 1 do livro , se clicar vai pra pagina 2.
rel='prev' = indica que ao clicar você vai voltar a pagina, exemplo: você esta na pagina 2 do livro, se clicar vai pra pagina 1
